    **1. Create src/components/OnboardingRedirect.tsx (per D-01, D-03):**

    ```typescript
    'use client';

    import { useEffect, useState } from 'react';
    import { useAuth } from '@clerk/nextjs';
    import { useRouter } from 'next/navigation';

    export function OnboardingRedirect() {
      const { isSignedIn, isLoaded } = useAuth();
      const router = useRouter();
      const [checked, setChecked] = useState(false);

      useEffect(() => {
        if (!isLoaded || !isSignedIn || checked) return;

        fetch('/api/onboarding')
          .then(r => r.json())
          .then(snapshot => {
            if (!snapshot.completed) {
              router.replace('/onboarding');
            }
            setChecked(true);
          })
          .catch(() => {
            // On error, don't redirect -- let user use the dashboard
            setChecked(true);
          });
      }, [isLoaded, isSignedIn, checked, router]);

      return null; // Invisible component, only performs redirect logic
    }
    ```

    Key behaviors:
    - Only fires for signed-in users (per D-05: unauthenticated users are unaffected).
    - Calls `GET /api/onboarding` which reads `user_onboarding_state.status` from DB.
    - If `snapshot.completed === false` (or no onboarding state exists), redirects to `/onboarding`.
    - If `snapshot.completed === true`, does nothing (user stays on dashboard).
    - Uses `router.replace` (not push) so `/onboarding` replaces history entry (per D-03: no intermediate screen).
    - The `checked` flag prevents re-fetching on re-renders.
    - On error, fails open (user stays on dashboard).
